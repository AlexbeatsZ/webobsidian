cmake_minimum_required(VERSION 3.22.1)
project(webobsidian)
get_filename_component(ANDROID_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${ANDROID_PROJECT}/.runtime/bin/${ANDROID_ABI}/libnode.so")
add_library(webobsidian SHARED runtime.cpp)
target_compile_features(webobsidian PRIVATE cxx_std_20)
target_include_directories(webobsidian PRIVATE "${ANDROID_PROJECT}/.runtime/include/node")
target_link_libraries(webobsidian node log)
target_link_options(webobsidian PRIVATE "-Wl,-z,max-page-size=16384")
